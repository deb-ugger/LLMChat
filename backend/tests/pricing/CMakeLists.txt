cmake_minimum_required(VERSION 3.16)
project(llmchat_pricing_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(pricing-tests main.cpp ../../src/PricingStore.cpp)
target_include_directories(pricing-tests PRIVATE ../../src ../../third_party)
if(WIN32)
    target_compile_definitions(pricing-tests PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(pricing-tests PRIVATE /utf-8 /EHsc)
endif()
enable_testing()
add_test(NAME pricing COMMAND pricing-tests)
